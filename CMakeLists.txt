cmake_minimum_required(VERSION 3.19)

project(qMdict
    VERSION 1.0.1
    DESCRIPTION "Fast offline MDict (.mdx/.mdd) dictionary reader"
    LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(QMDICT_BUILD_TESTS "Build the parser test harness" ON)
option(QMDICT_BUILD_ICON_TOOL "Build the icon generator (only needed to regenerate artwork)" OFF)

# Qt is the only dependency: zlib comes in through Qt's qUncompress, and the
# LZO and RIPEMD-128 codecs MDict needs are implemented in src/util.
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets)

qt_standard_project_setup()

# Speex decodes the .spx pronunciation clips most dictionaries ship. Only the
# decoder is compiled; the encoder and libspeexdsp are left out.
file(GLOB QMDICT_SPEEX_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/third_party/speex/libspeex/*.c)
add_library(qmdict_speex STATIC ${QMDICT_SPEEX_SOURCES})
target_include_directories(qmdict_speex PUBLIC third_party/speex/include)
target_include_directories(qmdict_speex PRIVATE third_party/speex third_party/speex/libspeex)
target_compile_definitions(qmdict_speex PRIVATE HAVE_CONFIG_H)
set_target_properties(qmdict_speex PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    # Vendored code; its warnings are not actionable for this project.
    target_compile_options(qmdict_speex PRIVATE -w -ffunction-sections -fdata-sections)
elseif(MSVC)
    target_compile_options(qmdict_speex PRIVATE /w)
endif()

# The MDict reader is a plain library so the test harness can link it without
# dragging in the GUI.
add_library(qmdict_core STATIC
    src/util/ripemd128.cpp
    src/util/lzo1x.cpp
    src/util/codec.cpp
    src/util/textdecode.cpp
    src/mdict/mdictfile.cpp
    src/mdict/dictionary.cpp
    src/mdict/library.cpp
    src/audio/oggstream.cpp
    src/audio/audiodecoder.cpp
    src/ui/darkcolours.cpp
    src/ui/htmlblocks.cpp
)

target_include_directories(qmdict_core PRIVATE third_party/minimp3)

# Qt6::Gui is here only for QColor in the dark-mode colour mapper; nothing in
# this library needs a GUI to be running.
target_link_libraries(qmdict_core PUBLIC Qt6::Core Qt6::Gui PRIVATE qmdict_speex)

if(NOT WIN32)
    # GNU libc provides iconv directly; elsewhere it is a separate library.
    find_library(ICONV_LIBRARY NAMES iconv)
    if(ICONV_LIBRARY)
        target_link_libraries(qmdict_core PUBLIC ${ICONV_LIBRARY})
    endif()
endif()

qt_add_executable(qMdict
    src/main.cpp
    src/ui/mainwindow.cpp
    src/ui/articleview.cpp
    src/ui/theme.cpp
    src/audio/audioplayer.cpp
)

target_link_libraries(qMdict PRIVATE qmdict_core Qt6::Widgets)

# Single source of truth for the version, so the About box and the release
# artefacts cannot drift apart.
target_compile_definitions(qMdict PRIVATE QMDICT_VERSION="${PROJECT_VERSION}")

# Window and taskbar icon. The artwork is generated by tools/make_icon.cpp.
qt_add_resources(qMdict "qmdict_icons"
    PREFIX "/"
    BASE resources
    FILES
        resources/icons/qmdict-16.png
        resources/icons/qmdict-24.png
        resources/icons/qmdict-32.png
        resources/icons/qmdict-48.png
        resources/icons/qmdict-64.png
        resources/icons/qmdict-128.png
        resources/icons/qmdict-256.png
)

if(WIN32)
    # PlaySound(), used to play decoded clips straight from memory.
    target_link_libraries(qMdict PRIVATE winmm)

    # Explorer reads the icon from the executable's resources, not from Qt's.
    set(QMDICT_ICO_PATH "${CMAKE_CURRENT_SOURCE_DIR}/resources/qmdict.ico")
    configure_file(resources/qmdict.rc.in "${CMAKE_CURRENT_BINARY_DIR}/qmdict.rc" @ONLY)
    target_sources(qMdict PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/qmdict.rc")
endif()

set_target_properties(qMdict PROPERTIES
    WIN32_EXECUTABLE ON
    MACOSX_BUNDLE OFF
)

# The packaged layout is bin/qMdict next to lib/, so the binary can find its
# bundled Qt without a wrapper script or LD_LIBRARY_PATH.
if(UNIX AND NOT APPLE)
    set_target_properties(qMdict PROPERTIES
        BUILD_RPATH "$ORIGIN/../lib"
        INSTALL_RPATH "$ORIGIN/../lib"
    )
endif()

# Trim the binary: no exceptions-heavy STL bloat, and drop unused sections.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(qmdict_core PRIVATE -ffunction-sections -fdata-sections)
    target_compile_options(qMdict PRIVATE -ffunction-sections -fdata-sections)
    if(NOT APPLE)
        target_link_options(qMdict PRIVATE -Wl,--gc-sections -Wl,--as-needed)
    endif()
endif()

if(QMDICT_BUILD_ICON_TOOL)
    add_executable(qmdict_make_icon tools/make_icon.cpp)
    target_link_libraries(qmdict_make_icon PRIVATE Qt6::Gui)
endif()

if(QMDICT_BUILD_TESTS)
    enable_testing()

    add_executable(qmdict_tests tests/test_mdict.cpp)
    # The Speex round-trip check needs the encoder, so the tests link the
    # vendored library directly rather than through qmdict_core.
    target_link_libraries(qmdict_tests PRIVATE qmdict_core qmdict_speex)
    add_test(NAME qmdict_tests COMMAND qmdict_tests)
endif()

include(GNUInstallDirs)
install(TARGETS qMdict RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
